    Use this in your app

    Import the reusable library and renderer. Parsing and saving run through the complete managed netDxf assembly; the scene contains the supported display primitives and per-entity issues.

    import { createRoslyn } from '../src/browser.js';
import { createNetDxf, createDxfRenderer } from '../src/dxf/index.js';

const compiler = await createRoslyn();
const dxf = await createNetDxf({ compiler });
const document = await dxf.load(new Uint8Array(await file.arrayBuffer()));
const renderer = await createDxfRenderer(canvas);
renderer.setScene(document.scene);
renderer.fit();
const bytes = await document.export({ binary: false });

// When finished: renderer.dispose(); await dxf.dispose(); compiler.dispose();
